function(monitoring_test name)
  add_executable(${name} ${name}.cpp)
  set_target_properties(${name} PROPERTIES AUTOMOC OFF AUTORCC OFF AUTOUIC OFF)
  target_link_libraries(${name} PRIVATE MonitoringCore)
  add_test(NAME ${name} COMMAND ${name})
  set_tests_properties(${name} PROPERTIES LABELS monitoring-unit TIMEOUT 20)
endfunction()
